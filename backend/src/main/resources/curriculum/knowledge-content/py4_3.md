## os 与 pathlib

### os 模块 — 操作系统接口

```python
import os

# 路径操作
os.getcwd()                 # 当前工作目录
os.chdir("/path/to/dir")    # 切换目录
os.listdir(".")             # 列出目录内容

# 创建/删除
os.mkdir("new_dir")         # 创建目录
os.makedirs("a/b/c")        # 递归创建
os.rmdir("empty_dir")       # 删除空目录
os.remove("file.txt")       # 删除文件

# 路径拼接
path = os.path.join("folder", "subfolder", "file.txt")

# 路径信息
os.path.exists(path)        # 路径是否存在
os.path.isfile(path)        # 是否是文件
os.path.isdir(path)         # 是否是目录
os.path.basename(path)      # 文件名
os.path.dirname(path)       # 目录名
os.path.splitext(path)      # 分离扩展名 → ("file", ".txt")
```

### pathlib — 现代路径操作

```python
from pathlib import Path

# 创建路径对象
p = Path("data/images/photo.jpg")

# 路径信息
p.name          # "photo.jpg"
p.stem          # "photo"
p.suffix        # ".jpg"
p.parent        # Path("data/images")
p.parent.parent # Path("data")

# 路径操作
p.exists()      # 是否存在
p.is_file()     # 是否是文件
p.is_dir()      # 是否是目录

# 拼接
data_dir = Path("data")
config_file = data_dir / "config" / "settings.yaml"

# 创建/删除
Path("new_dir").mkdir(exist_ok=True)
Path("a/b/c").mkdir(parents=True, exist_ok=True)
```

### 遍历文件

```python
# os.walk
import os
for root, dirs, files in os.walk("."):
    for file in files:
        print(os.path.join(root, file))

# pathlib — glob
from pathlib import Path
for py_file in Path("src").glob("**/*.py"):
    print(py_file)

# 过滤特定类型
for py_file in Path("src").rglob("*.py"):
    print(py_file)
```

### 文件信息

```python
from pathlib import Path
import os

p = Path("data.txt")
p.stat().st_size    # 文件大小（字节）
p.stat().st_mtime   # 修改时间

os.path.getsize("data.txt")      # 文件大小
os.path.getmtime("data.txt")     # 修改时间戳
```

### 小结

- `os.path` 是传统路径操作方式
- `pathlib.Path` 更现代、更面向对象
- 用 `/` 运算符拼接路径
- `glob()` 和 `rglob()` 查找文件
