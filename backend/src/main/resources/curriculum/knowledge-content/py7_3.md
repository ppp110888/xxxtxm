## 单元测试基础

### unittest 模块

```python
import unittest

# 被测试的函数
def add(a, b):
    return a + b

# 编写测试
class TestAdd(unittest.TestCase):
    def test_add_positive(self):
        self.assertEqual(add(2, 3), 5)

    def test_add_negative(self):
        self.assertEqual(add(-1, -1), -2)

    def test_add_zero(self):
        self.assertEqual(add(0, 5), 5)

    def test_add_raises(self):
        with self.assertRaises(TypeError):
            add("a", 1)

if __name__ == "__main__":
    unittest.main()
```

### 常用断言

```python
class TestExample(unittest.TestCase):
    def test_assertions(self):
        self.assertEqual(a, b)          # a == b
        self.assertNotEqual(a, b)       # a != b
        self.assertTrue(x)              # x is True
        self.assertFalse(x)             # x is False
        self.assertIs(a, b)             # a is b
        self.assertIsNone(x)            # x is None
        self.assertIsNotNone(x)         # x is not None
        self.assertIn(a, container)     # a in container
        self.assertNotIn(a, container)  # a not in container
        self.assertGreater(a, b)        # a > b
        self.assertLess(a, b)           # a < b
        self.assertAlmostEqual(a, b)    # 浮点数近似相等
        self.assertRaises(Error, func)  # 断言抛出异常
```

### setUp 和 tearDown

```python
class TestDatabase(unittest.TestCase):
    def setUp(self):
        """每个测试方法前执行"""
        self.data = [1, 2, 3]
        print("准备测试数据")

    def tearDown(self):
        """每个测试方法后执行"""
        self.data.clear()
        print("清理测试数据")

    def test_append(self):
        self.data.append(4)
        self.assertEqual(len(self.data), 4)

    def test_pop(self):
        self.data.pop()
        self.assertEqual(len(self.data), 2)

    @classmethod
    def setUpClass(cls):
        """所有测试前执行一次"""
        print("打开数据库连接")

    @classmethod
    def tearDownClass(cls):
        """所有测试后执行一次"""
        print("关闭数据库连接")
```

### pytest — 更简洁的测试

```python
# test_math.py
def add(a, b):
    return a + b

def test_add_positive():
    assert add(2, 3) == 5

def test_add_negative():
    assert add(-1, -1) == -2

def test_add_raises():
    import pytest
    with pytest.raises(TypeError):
        add("a", 1)
```

### 运行测试

```bash
# unittest
python -m unittest test_module.py
python -m unittest discover  # 自动发现

# pytest（需安装 pip install pytest）
pytest
pytest test_math.py -v     # verbose 模式
pytest --tb=short          # 简化回溯
```

### 小结

- `unittest` 是标准库测试框架
- 继承 `TestCase`，方法以 `test_` 开头
- `setUp`/`tearDown` 在每个测试前后运行
- `pytest` 语法更简洁，是业界主流
